"Content-Type": "application/json"
            },
            json=payload,
            timeout=config.REQUEST_TIMEOUT
        )


        if response.status_code >= 400:
            raise RuntimeError(
                f"Erro da API Groq ({response.status_code}): "
                f"{response.text[:400]}"
            )


        data = response.json()
        choices = data.get("choices", [])


        if not choices:
            raise RuntimeError(
                "O modelo não retornou uma resposta."
            )


        answer = choices[0]["message"]["content"]


        if not isinstance(answer, str) or not answer.strip():
            raise RuntimeError("Resposta vazia do modelo.")


        return answer.strip()

class Viva:


    def __init__(self):
        self.memory = memory
        self.cognition = CognitiveCore()
        self.language = LanguageEngine()


    def respond(
        self,
        conversation_id: str,
        user_text: str
    ) -> tuple[str, int]:


        started = time.time()


        relevant_memories = self.memory.recall(user_text)
        lessons = self.memory.feedback_lessons()


        history = self.memory.history(conversation_id)


        system_prompt = self.cognition.build_prompt(
            user_text,
            relevant_memories,
            lessons
        )


        messages = [
            {
                "role": "system",
                "content": system_prompt
            }
        ]


        # Recupera a continuidade da conversa.
        messages.extend(history[-config.MAX_HISTORY:])


        answer = self.language.generate(messages)


        # Persiste a resposta antes de devolver ao usuário.
        message_id = self.memory.add_message(
            conversation_id,
            "assistant",
            answer
        )


        # Registra uma experiência resumida.
        self.memory.remember(
            "Pergunta: "
            + user_text[:400]
            + "\nResposta: "
            + answer[:700],
            kind="experience",
            confidence=0.45
        )


        self.memory.metric(
            "response_seconds",
            time.time() - started,
            {"model": config.MODEL}
        )


        return answer, message_id

@app.post("/api/chat")
def chat():
    data = request.get_json(silent=True) or {}


    conversation_id = str(
        data.get("conversation_id", "")
    ).strip()


    user_text = str(
        data.get("message", "")
    ).strip()


    if not conversation_id or not user_text:
        return jsonify({
            "error": "Informe a conversa e a mensagem."
        }), 400


    if len(user_text) > config.MAX_MESSAGE_LENGTH:
        return jsonify({
            "error": "A mensagem excede o limite permitido."
        }), 413


    if not memory.conversation_exists(conversation_id):
        return jsonify({
            "error": "Conversa não encontrada."
        }), 404


    if not config.API_KEY:
        return jsonify({
            "error": "Configure GROQ_API_KEY no ambiente."
        }), 503


    try:
        memory.add_message(
            conversation_id,
            "user",
            user_text
        )


        answer, message_id = five.respond(
            conversation_id,
            user_text
        )


        return jsonify({
            "answer": answer,
            "message_id": message_id,
            "conversation_id": conversation_id
        })


    except requests.RequestException:
        app.logger.exception("Falha de comunicação com a API")


        return jsonify({
            "error": "Falha de conexão com o modelo de linguagem."
        }), 502


    except Exception:
        app.logger.exception("Falha no núcleo do 5")


        return jsonify({
            "error": "Ocorreu um erro interno ao gerar a resposta."
        }), 500




@app.post("/api/feedback")
def register_feedback():
    data = request.get_json(silent=True) or {}


    conversation_id = str(
        data.get("conversation_id", "")
    ).strip()


    try:
        rating = int(data.get("rating"))
    except (TypeError, ValueError):
        return jsonify({
            "error": "Avaliação inválida."
        }), 400


    if rating not in (-1, 0, 1):
        return jsonify({
            "error": "A avaliação deve ser -1, 0 ou 1."
        }), 400


    if not memory.conversation_exists(conversation_id):
        return jsonify({
            "error": "Conversa não encontrada."
        }), 404


    try:
        message_id = data.get("message_id")
        message_id = int(message_id) if message_id is not None else None
    except (TypeError, ValueError):
        message_id = None


    comment = str(data.get("comment", "")).strip()[:1000]


    memory.record_feedback(
        conversation_id,
        message_id,
        rating,
        comment
    )


    return jsonify({
        "status": "recorded",
        "reflection_proposals": memory.reflect()
    })

    data = request.get_json(silent=True) or {}

    kind = str(data.get("kind", "knowledge")).strip()

        return jsonify({

        }), 400

        }), 413


    memory_id = memory.remember(
        content,
        kind=kind[:50] or "knowledge",
        confidence=0.8
    )


    return jsonify({
        "status": "saved",
        "memory_id": memory_id
    })
